# Quickstart & Integration Guide

### 1. Installation

Install `@ruivalente99/bibliotheca` in your Next.js or React 19 application using Bun:

```bash
# Using Bun (recommended)
bun add @ruivalente99/bibliotheca

# Or link locally during development
bun add @ruivalente99/bibliotheca@file:../bibliotheca
```

---

### 2. Tailwind CSS v4 Configuration

Bibliotheca is styled with modern Tailwind CSS v4 utility classes and CSS variables. Simply import the styles and point Tailwind to Bibliotheca components:

```css
/* app/globals.css */
@import "tailwindcss";
@import "@ruivalente99/bibliotheca/styles.css";

/* Point Tailwind to Bibliotheca source files */
@source "../node_modules/@ruivalente99/bibliotheca";
```

---

### 3. Application Root Providers

Wrap your root layout with `ThemeProvider` and `ToastProvider`:

```tsx
// app/layout.tsx
import { ThemeProvider, ToastProvider } from "@ruivalente99/bibliotheca/ui";
import "./globals.css";

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en" suppressHydrationWarning>
      <body>
        <ThemeProvider defaultTheme="system" defaultAccent="amber">
          <ToastProvider autoDismissMs={4000}>
            {children}
          </ToastProvider>
        </ThemeProvider>
      </body>
    </html>
  );
}
```

### 4. Assembling a Full Studio Experience

Here is a full minimal example combining the header, editor panes, canvas viewport, and PDF export:

```tsx
"use client";

import { useState, useRef } from "react";
import {
  SplitEditorLayout,
  BuilderHeader,
  SectionCard,
  useSectionSync,
} from "@ruivalente99/bibliotheca/editor";
import { PreviewViewport } from "@ruivalente99/bibliotheca/preview";
import { Button, Input, useToast } from "@ruivalente99/bibliotheca/ui";
import { exportNodeToPdf } from "@ruivalente99/bibliotheca/export";

export default function DocumentStudio() {
  const { showToast } = useToast();
  const { highlightedSectionId, handleSelectSection } = useSectionSync();
  const docRef = useRef<HTMLDivElement>(null);
  const [fullName, setFullName] = useState("Alexandre Valente");

  const handleExport = async () => {
    if (!docRef.current) return;
    await exportNodeToPdf(docRef.current, { filename: "document.pdf" });
    showToast("PDF downloaded successfully.", "success");
  };

  const form = (
    <div className="builder-form-pane p-6 space-y-6">
      <SectionCard
        id="section-personal"
        title="Personal Information"
        highlighted={highlightedSectionId === "personal"}
      >
        <Input
          label="Full Name"
          value={fullName}
          onChange={(e) => setFullName(e.target.value)}
          placeholder="Enter legal name"
        />
      </SectionCard>
    </div>
  );

  const preview = (
    <PreviewViewport
      docWidth={794}
      docHeight={1123}
      toolbarActions={
        <Button variant="primary" size="sm" onClick={handleExport}>
          Download PDF
        </Button>
      }
    >
      <div
        ref={docRef}
        onClick={() => handleSelectSection("personal")}
        className="w-[794px] h-[1123px] bg-white shadow-2xl p-12 cursor-pointer select-none"
      >
        <h1 className="text-2xl font-bold text-stone-900">{fullName}</h1>
        <p className="text-sm text-stone-600 mt-2">
          Click any document block to highlight the corresponding form section.
        </p>
      </div>
    </PreviewViewport>
  );

  return (
    <SplitEditorLayout
      header={
        <BuilderHeader
          title="Document Studio"
          subtitle="Next-generation publishing workbench"
        />
      }
      formPane={form}
      previewPane={preview}
    />
  );
}
```
